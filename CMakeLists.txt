cmake_minimum_required(VERSION 3.16)
project(AirXpress CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(CORE_SOURCES
  src/world.cpp src/aircraft.cpp src/career.cpp)
set(GAME_SOURCES
  ${CORE_SOURCES} src/game.cpp src/game_ui.cpp src/renderer.cpp src/gl.cpp src/audio.cpp)

if(WIN32)
  if(MINGW)
    enable_language(RC)
  endif()
  add_executable(AirXpress WIN32 ${GAME_SOURCES} src/platform_win32.cpp src/radio_win.cpp assets/app.rc)
  target_link_libraries(AirXpress PRIVATE opengl32 gdi32 winmm mfplay mfplat mfuuid ole32 uuid)
  if(MSVC)
    set_property(TARGET AirXpress PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(AirXpress PRIVATE /O2 /fp:fast /W3 /wd4244 /wd4305 /wd4267)
    target_compile_definitions(AirXpress PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(AirXpress PRIVATE -O2)
    target_link_options(AirXpress PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()

# Headless tests (flight model + campaign progression validation)
add_executable(flight_test tests/flight_test.cpp ${CORE_SOURCES})
add_executable(progression_test tests/progression_test.cpp ${CORE_SOURCES})
if(MSVC)
  target_compile_definitions(flight_test PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_definitions(progression_test PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
enable_testing()
add_test(NAME flight_model COMMAND flight_test)
add_test(NAME campaign_progression COMMAND progression_test)
