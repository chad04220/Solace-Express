cmake_minimum_required(VERSION 3.16)
project(SolaceExpress CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
# Linux CI: every test under AddressSanitizer + UndefinedBehaviorSanitizer (-DSANITIZE=ON)
option(SANITIZE "Build with -fsanitize=address,undefined" OFF)
if(SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined -O1)
  add_link_options(-fsanitize=address,undefined)
endif()

# GLSL sources (src/shaders/*.glsl) embedded as C++ constants in gen/shaders_gen.h at build time
set(SHADER_GEN ${CMAKE_BINARY_DIR}/gen/shaders_gen.h)
file(GLOB SHADER_SRCS CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/shaders/*.glsl)
add_custom_command(OUTPUT ${SHADER_GEN}
  COMMAND ${CMAKE_COMMAND} -DSRC_DIR=${CMAKE_SOURCE_DIR}/src/shaders -DOUT=${SHADER_GEN} -P ${CMAKE_SOURCE_DIR}/tools/embed_shaders.cmake
  DEPENDS ${SHADER_SRCS} ${CMAKE_SOURCE_DIR}/tools/embed_shaders.cmake
  COMMENT "Embedding GLSL shaders")
add_custom_target(shaders_gen DEPENDS ${SHADER_GEN})
include_directories(${CMAKE_BINARY_DIR}/gen)

set(CORE_SOURCES
  src/world.cpp src/scenery.cpp src/entities.cpp src/airport_scenery.cpp src/aircraft.cpp src/aircraft_perf.cpp src/aircraft_stunt.cpp src/aero.cpp src/career.cpp src/weather.cpp)
set(GAME_SOURCES
  ${CORE_SOURCES} src/traffic.cpp src/game.cpp src/game_ui.cpp src/game_research_ui.cpp src/game_wraith.cpp src/renderer.cpp src/materials.cpp src/raster_renderer.cpp src/terrain_mesh.cpp src/aircraft_mesh.cpp src/entity_render.cpp src/aircraft_hull.cpp src/camera_feeds.cpp src/atc.cpp src/entity_mesh.cpp src/gl.cpp src/audio.cpp src/models.cpp)

if(WIN32)
  if(MINGW)
    enable_language(RC)
  endif()
  add_executable(SolaceExpress WIN32 ${GAME_SOURCES} src/platform_win32.cpp src/radio_win.cpp assets/app.rc)
  target_link_libraries(SolaceExpress PRIVATE opengl32 gdi32 winmm mfplay mfplat mfuuid ole32 uuid)
  if(MSVC)
    set_property(TARGET SolaceExpress PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(SolaceExpress PRIVATE /O2 /fp:fast /W3 /wd4244 /wd4305 /wd4267)
    target_compile_definitions(SolaceExpress PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(SolaceExpress PRIVATE -O2)
    target_link_options(SolaceExpress PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()

# Development render harness on Linux (Xvfb + Mesa): test frames, and a render server (render_harness serve) that keeps
# everything loaded between shots. Not built by default: cmake --build build --target render_harness
if(UNIX AND NOT APPLE)
  add_executable(render_harness EXCLUDE_FROM_ALL tests/render_harness.cpp ${GAME_SOURCES} src/radio_stub.cpp)
  add_dependencies(render_harness shaders_gen)
  target_link_libraries(render_harness PRIVATE ${CMAKE_DL_LIBS} X11 pthread)
  # the aircraft fields' geometry probes (headrest clearance, end-cap seams, the XR-40's touchpads) on Mesa/EGL, across
  # the whole roster: aircraft_visual_test OUTPUT_DIR [W H] [first last]
  add_executable(aircraft_visual_test EXCLUDE_FROM_ALL tests/aircraft_visual_test.cpp ${CORE_SOURCES} src/models.cpp src/gl.cpp)
  add_dependencies(aircraft_visual_test shaders_gen)
  target_link_libraries(aircraft_visual_test PRIVATE ${CMAKE_DL_LIBS} pthread)
endif()
# the procedural material layers as images, for tools/pack_materials.py: material_dump OUTPUT_DIR

target_include_directories(material_dump PRIVATE src)

# everything that includes the shader headers waits for the generated constants
foreach(t SolaceExpress gameplay_test encounter_timing_test shader_keyword_test shader_check terrain_shadow_test)
  if(TARGET ${t})
    add_dependencies(${t} shaders_gen)
  endif()
endforeach()
