# MSVC's Visual Studio generator uses source.obj; Ninja/Make preserve source.cpp.o
# (or source.cpp.obj). Exclude the strict object before linking its fast-math copy.
# Check both naming styles at configure time, including similarly named objects
# that must remain, so a Linux-only build cannot hide another MSVC duplicate link.
foreach(source career hive_world_collision)
  set(${source}_object_filter "(^|/)${source}([.]cpp)?[.](o|obj)$")
  set(object_filter_probe "core.dir/${source}.cpp.o;core.dir/${source}.cpp.obj;core.dir/Release/${source}.obj;${source}.obj;core.dir/other_${source}.obj;core.dir/${source}_test.cpp.o")
  list(FILTER object_filter_probe EXCLUDE REGEX "${${source}_object_filter}")
  if(NOT object_filter_probe STREQUAL "core.dir/other_${source}.obj;core.dir/${source}_test.cpp.o")
    message(FATAL_ERROR "Fast-math object exclusion failed for ${source}: ${object_filter_probe}")
  endif()
endforeach()

add_executable(steam_test tests/steam_test.cpp src/steam.cpp ${CORE_OBJS})
target_compile_definitions(steam_test PRIVATE SOMNIUM_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
if(MSVC)
  target_compile_definitions(steam_test PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
if(WIN32)
  add_library(steam_fake SHARED tests/steam_fake_dll.cpp)
  set_target_properties(steam_fake PROPERTIES OUTPUT_NAME steam_api64 PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/steam_fake$<0:>")
  add_dependencies(steam_test steam_fake)
  add_test(NAME steam COMMAND steam_test $<TARGET_FILE_DIR:steam_fake>)
else()   # (the same loader against the stand-in built as a shared object: under the sanitizers too)
  add_library(steam_fake SHARED tests/steam_fake_dll.cpp)
  set_target_properties(steam_fake PROPERTIES OUTPUT_NAME steam_api64 PREFIX "" SUFFIX ".so" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/steam_fake")
  target_compile_definitions(steam_test PRIVATE SOMNIUM_STEAM_DLOPEN=1)
  target_link_libraries(steam_test PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(steam_test steam_fake)
  add_test(NAME steam COMMAND steam_test $<TARGET_FILE_DIR:steam_fake>)
endif()
